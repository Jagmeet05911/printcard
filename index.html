function makeOutput(s, p) { const dpi = 300; let mmW = p.w; let mmH = p.h; const orientation = $('orientation').value; if (orientation === 'landscape' && mmH > mmW) { [mmW, mmH] = [mmH, mmW]; } if (orientation === 'portrait' && mmW > mmH) { [mmW, mmH] = [mmH, mmW]; } const outW = Math.round((mmW / 25.4) * dpi); const outH = Math.round((mmH / 25.4) * dpi); const output = document.createElement('canvas'); output.width = outW; output.height = outH; const ctx = output.getContext('2d'); ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, outW, outH); const img = s.img; const iw = img.naturalWidth || img.width; const ih = img.naturalHeight || img.height; const crop = s.crop || { x: 0, y: 0, w: 1, h: 1 }; const sx = Math.max(0, Math.floor(crop.x * iw)); const sy = Math.max(0, Math.floor(crop.y * ih)); const sw = Math.max(1, Math.min(iw - sx, Math.round(crop.w * iw))); const sh = Math.max(1, Math.min(ih - sy, Math.round(crop.h * ih))); // 1. Crop the selected part of the original image const cropped = document.createElement('canvas'); cropped.width = sw; cropped.height = sh; const cropCtx = cropped.getContext('2d'); cropCtx.drawImage( img, sx, sy, sw, sh, 0, 0, sw, sh ); // 2. Rotate and mirror the cropped image const rotation = ((s.rotation || 0) % 360 + 360) % 360; const sideways = rotation === 90 || rotation === 270; const rotated = document.createElement('canvas'); rotated.width = sideways ? sh : sw; rotated.height = sideways ? sw : sh; const rctx = rotated.getContext('2d'); rctx.translate(rotated.width / 2, rotated.height / 2); rctx.rotate(rotation * Math.PI / 180); rctx.scale(s.mirror ? -1 : 1, 1); rctx.drawImage(cropped, -sw / 2, -sh / 2); // 3. Fit the image inside the card with the selected border const border = Math.max( 0, Math.min(10, Number($('border').value) || 0) ); const inset = Math.round((border / 25.4) * dpi); const maxW = Math.max(1, outW - inset * 2); const maxH = Math.max(1, outH - inset * 2); const ratio = rotated.width / rotated.height; let drawW; let drawH; if (maxW / maxH > ratio) { drawH = maxH; drawW = Math.round(drawH * ratio); } else { drawW = maxW; drawH = Math.round(drawW / ratio); } ctx.drawImage( rotated, (outW - drawW) / 2, (outH - drawH) / 2, drawW, drawH ); return { canvas: output, w: mmW, h: mmH }; }
